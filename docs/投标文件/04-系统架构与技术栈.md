class LLMRouter:
    """大模型路由器"""
    
    def __init__(self):
        self.models = {
            'gpt5': OpenAIClient(model='gpt-5'),

### 4.4.2 向量数据库

**Pinecone / Milvus：**
```python
# 向量化与检索流程
class VectorSearch:
    def __init__(self):
        self.encoder = SentenceTransformer('paraphrase-multilingual-MiniLM-L12-v2')
        self.index = PineconeIndex(dimension=384)
    
    def add_documents(self, documents):
        """添加文档到向量库"""
        for doc in documents:
            # 文档向量化
            vector = self.encoder.encode(doc.content)
            # 存入向量库
            self.index.upsert([{
                'id': doc.id,
                'values': vector,
                'metadata': doc.metadata
            }])
    
    def search(self, query, top_k=5):
        """语义搜索"""
        query_vector = self.encoder.encode(query)
        results = self.index.query(
            vector=query_vector,
            top_k=top_k,
            include_metadata=True
        )
        return results
```

WHERE c1 <> c2
RETURN c2.name, count(k) as common_knowledge
ORDER BY common_knowledge DESC
LIMIT 5
